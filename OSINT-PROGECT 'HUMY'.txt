class NetworkManager {
constructor(app) {
this.app = app;
this.peer = null;
this.connections = new Map();
this.enabled = false;
this.myId = null;
}

toggle() {
if (this.enabled) this.disable();
else this.enable();
}

enable() {
if (this.enabled) return;
this.setStatus('connecting', 'Подключение...');
const id = 'osint-' + Math.random().toString(36).substr(2, 9);
try {
this.peer = new Peer(id, {
debug: 1,
config: {
iceServers: [
{ urls: 'stun:stun.l.google.com:19302' },
{ urls: 'stun:stun1.l.google.com:19302' }
 ]
}
});

this.peer.on('open', (assignedId) => {
this.myId = assignedId;
this.enabled = true;
document.getElementById('myPeerId').textContent = assignedId;
document.getElementById('netToggleText').textContent = 'Отключить сеть';
this.setStatus('online', 'Онлайн');
this.app.flash('Сеть активна. Ваш ID: ' + assignedId);
});

this.peer.on('connection', (conn) => this._setupConnection(conn));

this.peer.on('error', (err) => {
console.error('Peer error:', err);
if (err.type === 'unavailable-id') {
this.app.flash('ID занят, пробуем другой...');
this.disable();
setTimeout(() => this.enable(), 500);
} else {
this.setStatus('', 'Ошибка: ' + err.type);
this.app.flash('Ошибка сети: ' + err.type);
}
});

this.peer.on('disconnected', () => {
this.setStatus('', 'Отключён');
});
} catch (e) {
this.app.flash('Не удалось инициализировать сеть');
this.setStatus('', 'Ошибка');
}
}

disable() {
if (this.peer) {
this.peer.destroy();
this.peer = null;
}
this.connections.clear();
this.enabled = false;
this.myId = null;
document.getElementById('myPeerId').textContent = '—';
document.getElementById('netToggleText').textContent = 'Включить сеть';
this.setStatus('', 'Оффлайн');
this._renderPeers();
}

connect() {
if (!this.enabled) { this.app.flash('Сначала включите сеть'); return; }
const targetId = document.getElementById('peerIdInput').value.trim();
if (!targetId) { this.app.flash('Введите ID пира'); return; }
if (targetId === this.myId) { this.app.flash('Нельзя подключиться к самому себе'); return; }
if (this.connections.has(targetId)) { this.app.flash('Уже подключены'); return; }
try {
const conn = this.peer.connect(targetId, { reliable: true });
this._setupConnection(conn);
this.app.flash('Подключение к ' + targetId + '...');
} catch (e) {
this.app.flash('Ошибка подключения');
}
}

_setupConnection(conn) {
conn.on('open', () => {
this.connections.set(conn.peer, conn);
this._renderPeers();
this.app.flash('Подключён: ' + conn.peer);
conn.send({ type: 'sync', data: this.app.serialize() });
});
conn.on('data', (data) => this._handleMessage(data, conn.peer));
conn.on('close', () => {
this.connections.delete(conn.peer);
this._renderPeers();
this.app.flash('Отключён: ' + conn.peer);
});
conn.on('error', (err) => {
console.error('Connection error:', err);
this.connections.delete(conn.peer);
this._renderPeers();
});
}

_handleMessage(msg, fromPeer) {
this.app._remoteChange = true;
try {
switch (msg.type) {
case 'sync': this.app.deserialize(msg.data); break;
case 'node:add': this.app._addNodeRemote(msg.data); break;
case 'node:move': this.app._moveNodeRemote(msg.data.id, msg.data.x, msg.data.y); break;
case 'node:delete': this.app._deleteNodeRemote(msg.data.id); break;
case 'node:edit': this.app._editNodeRemote(msg.data); break;
case 'edge:add': this.app._addEdgeRemote(msg.data); break;
case 'edge:delete': this.app._deleteEdgeRemote(msg.data); break;
}
} catch (e) { console.error('Handle message error:', e); }
this.app._remoteChange = false;
}

broadcast(msg, excludePeer = null) {
if (!this.enabled) return;
for (const [peerId, conn] of this.connections) {
if (peerId === excludePeer) continue;
try { if (conn.open) conn.send(msg); }
catch (e) { console.error('Broadcast error:', e); }
}
}

disconnectPeer(peerId) {
const conn = this.connections.get(peerId);
if (conn) {
conn.close();
this.connections.delete(peerId);
this._renderPeers();
}
}

_renderPeers() {
const list = document.getElementById('peersList');
if (this.connections.size === 0) {
list.innerHTML = '<div style="color:#666;font-size:10px;padding:4px;">Нет подключений</div>';
return;
}
list.innerHTML = [...this.connections.keys()].map(id => &lt;div class="peer-item"&gt; &lt;span class="peer-id"&gt;● ${id}</span>
<button class="peer-disconnect" onclick="app.net.disconnectPeer('${id}')" title="Отключить"&gt;✕&lt;/button&gt; &lt;/div&gt;).join('');
}

setStatus(state, text) {
const dot = document.getElementById('netStatusDot');
const txt = document.getElementById('netStatusText');
dot.className = 'status-dot' + (state ? ' ' + state : '');
txt.textContent = text;
}

copyId() {
if (!this.myId) { this.app.flash('Сначала включите сеть'); return; }
navigator.clipboard.writeText(this.myId).then(() => {
this.app.flash('ID скопирован');
}).catch(() => {
const ta = document.createElement('textarea');
ta.value = this.myId;
document.body.appendChild(ta);
ta.select();
document.execCommand('copy');
ta.remove();
this.app.flash('ID скопирован');
});
}
}

this.nodes = new Map();
this.edges = [];
this.nextId = 1;

this.offset = { x: 0, y: 0 };
this.scale = 1;

this.selectedNode = null;
this.dragNode = null;
this.dragOffset = { x: 0, y: 0 };
this.isPanning = false;
this.panStart = { x: 0, y: 0 };
this.connecting = null;
this.mouse = { x: 0, y: 0 };

this._remoteChange = false;
this._lastBroadcastPos = new Map();
this._saveTimer = null;

this.net = new NetworkManager(this);
this.init();
this._restoreFromStorage();
}

init() {
this.resize();
window.addEventListener('resize', () => this.resize());

const list = document.getElementById('entityList');
for (const [key, t] of Object.entries(ENTITY_TYPES)) {

const rect = this.canvasWrap.getBoundingClientRect();

}

// ---- Автосохранение ----
_scheduleSave() {
if (this._saveTimer) clearTimeout(this._saveTimer);
this._saveTimer = setTimeout(() => this._saveNow(), 400);
}

_saveNow() {
try {
const data = {
graph: this.serialize(),
view: { offset: this.offset, scale: this.scale }
};
localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
} catch (e) { console.warn('Save failed:', e); }
}

_restoreFromStorage() {
try {
const raw = localStorage.getItem(STORAGE_KEY);
if (!raw) return;
const data = JSON.parse(raw);
if (data.graph) this.deserialize(data.graph);
if (data.view) {
this.offset = data.view.offset || { x: 0, y: 0 };
this.scale = data.view.scale || 1;
this.updateZoomLabel();
}
this.updateAllNodePositions();
this.render();
} catch (e) { console.warn('Restore failed:', e); }
}

resize() {
const rect = this.canvasWrap.getBoundingClientRect();
const dpr = window.devicePixelRatio || 1;
this.canvas.width = rect.width * dpr;
this.canvas.height = rect.height * dpr;
this.canvas.style.width = rect.width + 'px';
this.canvas.style.height = rect.height + 'px';
this.ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
this.render();
}

screenToWorld(sx, sy) {
const rect = this.canvasWrap.getBoundingClientRect();
return {
x: (sx - rect.left - this.offset.x) / this.scale,
y: (sy - rect.top - this.offset.y) / this.scale
};
}

// Конвертирует мировые координаты узла в ЭКРАННЫЕ (CSS-пиксели относительно canvas-wrap)
worldToNodeScreen(wx, wy) {
return {
x: wx * this.scale + this.offset.x,
y: wy * this.scale + this.offset.y
};
}

// Обновляет DOM-позицию одного узла
updateNodeScreenPosition(node) {
if (!node.el) return;
const s = this.worldToNodeScreen(node.x, node.y);
node.el.style.left = s.x + 'px';
node.el.style.top = s.y + 'px';
}

// Обновляет позиции ВСЕХ узлов (нужно при pan/zoom)
updateAllNodePositions() {
for (const node of this.nodes.values()) {
this.updateNodeScreenPosition(node);
}
}

// ---- Nodes ----
addNode(type, x, y, label = '', broadcast = true) {
const id = this.nextId++;
const node = { id, type, x, y, label: label || ${ENTITY_TYPES[type].label} #${id} };
this.nodes.set(id, node);
this.createNodeElement(node);
this.updateStats();
this.render();
if (broadcast && !this._remoteChange) {
this.net.broadcast({ type: 'node:add', data: node });
}
this._scheduleSave();
return node;
}

_addNodeRemote(data) {
if (this.nodes.has(data.id)) return;
const node = { ...data };
this.nodes.set(node.id, node);
if (node.id >= this.nextId) this.nextId = node.id + 1;
this.createNodeElement(node);
this.updateStats();
this.render();
this._scheduleSave();
}

updateNodePosition(node, broadcast = true) {
if (!node.el) return;
this.updateNodeScreenPosition(node);
if (broadcast && !this._remoteChange) {
const now = Date.now();
const last = this._lastBroadcastPos.get(node.id) || 0;
if (now - last > 50) {
this._lastBroadcastPos.set(node.id, now);
this.net.broadcast({ type: 'node:move', data: { id: node.id, x: node.x, y: node.y } });
}
}
}

_moveNodeRemote(id, x, y) {
const node = this.nodes.get(id);
if (!node) return;
node.x = x;
node.y = y;
this.updateNodePosition(node, false);
this.render();
this._scheduleSave();
}

selectNode(id) {
if (this.selectedNode && this.selectedNode.el) {
this.selectedNode.el.classList.remove('selected');
}
this.selectedNode = id != null ? this.nodes.get(id) : null;
if (this.selectedNode && this.selectedNode.el) {
this.selectedNode.el.classList.add('selected');
}
}

deleteNode(id, broadcast = true) {
const node = this.nodes.get(id);
if (!node) return;
if (node.el) node.el.remove();
this.nodes.delete(id);
const removedEdges = this.edges.filter(e => e.from === id || e.to === id);
this.edges = this.edges.filter(e => e.from !== id && e.to !== id);
if (this.selectedNode && this.selectedNode.id === id) this.selectedNode = null;
this.updateStats();
this.render();
if (broadcast && !this._remoteChange) {
this.net.broadcast({ type: 'node:delete', data: { id } });
for (const edge of removedEdges) {
this.net.broadcast({ type: 'edge:delete', data: edge });
}
}
this._scheduleSave();
}

_deleteNodeRemote(id) {
const node = this.nodes.get(id);
if (!node) return;
if (node.el) node.el.remove();
this.nodes.delete(id);
this.edges = this.edges.filter(e => e.from !== id && e.to !== id);
if (this.selectedNode && this.selectedNode.id === id) this.selectedNode = null;
this.updateStats();
this.render();
this._scheduleSave();
}

// ---- Edges ----
addEdge(fromId, toId, label = '', broadcast = true) {
if (fromId === toId) return;
if (this.edges.some(e => e.from === fromId && e.to === toId)) return;
const edge = { from: fromId, to: toId, label };
this.edges.push(edge);
this.updateStats();
this.render();
if (broadcast && !this._remoteChange) {
this.net.broadcast({ type: 'edge:add', data: edge });
}
this._scheduleSave();
}

_addEdgeRemote(data) {
if (data.from === data.to) return;
if (this.edges.some(e => e.from === data.from && e.to === data.to)) return;
this.edges.push({ ...data });
this.updateStats();
this.render();
this._scheduleSave();
}

_deleteEdgeRemote(data) {
this.edges = this.edges.filter(e => !(e.from === data.from && e.to === data.to));
this.updateStats();
this.render();
this._scheduleSave();
}

// ---- Rendering ----
render() {
const ctx = this.ctx;
const w = this.canvas.clientWidth;
const h = this.canvas.clientHeight;
ctx.clearRect(0, 0, w, h);

ctx.save();
ctx.translate(this.offset.x, this.offset.y);
ctx.scale(this.scale, this.scale);

for (const edge of this.edges) this.drawEdge(edge);

if (this.connecting) {
const from = this.nodes.get(this.connecting.fromId);
if (from) {
const mouseWorld = this.screenToWorld(this.mouse.x, this.mouse.y);
this.drawCurve(from.x, from.y, mouseWorld.x, mouseWorld.y, false, '', true);
}
}

ctx.restore();
}

drawEdge(edge) {
const a = this.nodes.get(edge.from);
const b = this.nodes.get(edge.to);
if (!a || !b) return;
this.drawCurve(a.x, a.y, b.x, b.y, false, edge.label, false);
}

/**
• Стрелка как в Obsidian: линия подходит вплотную к краю иконки (круга).
• Узлы хранятся в МИРОВЫХ координатах. Canvas имеет ctx.scale(this.scale).
• Визуальный радиус иконки в ЭКРАННЫХ пикселях = NODE_VISUAL_RADIUS (28px).
• Значит в МИРОВЫХ координатах радиус = NODE_VISUAL_RADIUS / this.scale.
• isPreview — для превью при создании связи (второй конец — курсор,
• там нет узла, поэтому к концу не обрезаем).
/
drawCurve(x1, y1, x2, y2, dashed = false, label = '', isPreview = false) {
const ctx = this.ctx;
const dx = x2 - x1;
const dy = y2 - y1;
const dist = Math.sqrt(dxdx + dy*dy);
if (dist < 1) return;

const ux = dx / dist;
const uy = dy / dist;

// Радиус в МИРОВЫХ координатах (чтобы соответствовать визуальному радиусу иконки)
const r = NODE_VISUAL_RADIUS / this.scale;

// Начало стрелки — на КРАЮ исходного узла
const startX = x1 + ux * r;
const startY = y1 + uy * r;

// Конец стрелки:
// - для обычной связи — на КРАЮ целевого узла (чтобы наконечник упирался в иконку)
// - для превью — прямо в точку курсора
let endX, endY;
if (isPreview) {
endX = x2;
endY = y2;
} else {
// Оставляем место для наконечника стрелки — чуть раньше края,
// чтобы остриё наконечника касалось края иконки
const arrowSize = 11 / this.scale; // в мировых
endX = x2 - ux * (r + arrowSize * 0.15);
endY = y2 - uy * (r + arrowSize * 0.15);
}

const sdx = endX - startX;
const sdy = endY - startY;
const sdist = Math.sqrt(sdxsdx + sdysdy);
if (sdist < 1) return;

const nx = -sdy / sdist;
const ny = sdx / sdist;

const curvature = Math.min(sdist * 0.3, 100 / this.scale);
const bend = curvature * 0.35;

const cx1 = startX + sdx * 0.33 + nx * bend;
const cy1 = startY + sdy * 0.33 + ny * bend;
const cx2 = startX + sdx * 0.66 + nx * bend;
const cy2 = startY + sdy * 0.66 + ny * bend;

// Градиент линии
const grad = ctx.createLinearGradient(startX, startY, endX, endY);
grad.addColorStop(0, '#888');
grad.addColorStop(1, '#fff');

ctx.strokeStyle = grad;
ctx.lineWidth = 2 / this.scale;
ctx.lineCap = 'round';
ctx.lineJoin = 'round';
ctx.setLineDash([]);

ctx.beginPath();
ctx.moveTo(startX, startY);
ctx.bezierCurveTo(cx1, cy1, cx2, cy2, endX, endY);
ctx.stroke();

// Наконечник стрелки (только для обычных связей, не для превью)
if (!isPreview) {
// Точка, куда упрётся остриё — ровно на краю целевого узла
const tipX = x2 - ux * r;
const tipY = y2 - uy * r;

// Касательная в конце кривой (для ориентации наконечника)
const t = 0.96;
const u = 1 - t;
const tx = 3uu*(cx1-startX) + 6ut*(cx2-cx1) + 3tt*(endX-cx2);
const ty = 3uu*(cy1-startY) + 6ut*(cy2-cy1) + 3tt*(endY-cy2);
const angle = Math.atan2(ty, tx);
const arrowSize = 11 / this.scale;

ctx.fillStyle = '#fff';
ctx.strokeStyle = '#000';
ctx.lineWidth = 1 / this.scale;
ctx.beginPath();
ctx.moveTo(tipX, tipY);
ctx.lineTo(tipX - arrowSize * Math.cos(angle - Math.PI/7),
tipY - arrowSize * Math.sin(angle - Math.PI/7));
ctx.lineTo(tipX - arrowSize * Math.cos(angle + Math.PI/7),
tipY - arrowSize * Math.sin(angle + Math.PI/7));
ctx.closePath();
ctx.fill();
ctx.stroke();
}

// Метка связи
if (label) {
const mx = (startX + endX) / 2 + nx * bend * 0.5;
const my = (startY + endY) / 2 + ny * bend * 0.5;
const fontSize = 11 / this.scale;
ctx.font = ${fontSize}px Segoe UI`;
const tw = ctx.measureText(label).width;
const pad = 5 / this.scale;
ctx.fillStyle = '#000';
ctx.fillRect(mx - tw/2 - pad, my - fontSize/2 - pad/2, tw + pad2, fontSize + pad);
ctx.strokeStyle = '#fff';
ctx.lineWidth = 1 / this.scale;
ctx.strokeRect(mx - tw/2 - pad, my - fontSize/2 - pad/2, tw + pad2, fontSize + pad);
ctx.fillStyle = '#fff';
ctx.textAlign = 'center';
ctx.textBaseline = 'middle';
ctx.fillText(label, mx, my);
}
}

// ---- Mouse Events ----
onMouseDown(e) {
if (e.target.closest('.node')) return;
if (e.button === 0) {
this.isPanning = true;
this.panStart = { x: e.clientX - this.offset.x, y: e.clientY - this.offset.y };
this.canvasWrap.style.cursor = 'grabbing';
this.selectNode(null);
}
}

onMouseMove(e) {
this.mouse.x = e.clientX;
this.mouse.y = e.clientY;

if (this.dragNode) {
const rect = this.canvasWrap.getBoundingClientRect();
// Курсор в экранных координатах -> мировые
const sx = e.clientX - rect.left;
const sy = e.clientY - rect.top;
this.dragNode.x = (sx - this.offset.x - this.dragOffset.x) / this.scale;
this.dragNode.y = (sy - this.offset.y - this.dragOffset.y) / this.scale;
this.updateNodePosition(this.dragNode);
this.render();
} else if (this.isPanning) {
this.offset.x = e.clientX - this.panStart.x;
this.offset.y = e.clientY - this.panStart.y;
this.updateAllNodePositions();
this.render();
} else if (this.connecting) {
this.render();
}
}

onMouseUp(e) {
if (this.dragNode) {
this.net.broadcast({ type: 'node:move', data: { id: this.dragNode.id, x: this.dragNode.x, y: this.dragNode.y } });
this.dragNode = null;
this._scheduleSave();
}
if (this.isPanning) {
this.isPanning = false;
this.canvasWrap.style.cursor = 'default';
this._scheduleSave();
}
if (this.connecting) {
const world = this.screenToWorld(e.clientX, e.clientY);
let target = null;
const hitRadius = NODE_VISUAL_RADIUS / this.scale + 5 / this.scale;
for (const n of this.nodes.values()) {
const dx = n.x - world.x, dy = n.y - world.y;
if (dxdx + dydy < hitRadius*hitRadius) { target = n; break; }
}
if (target && target.id !== this.connecting.fromId) {
this.promptEdgeLabel(this.connecting.fromId, target.id);
}
this.connecting = null;
this.render();
}
}

// ---- Touch Events ----
onTouchStart(e) {
if (e.target.closest('.node')) return;
if (e.touches.length === 1) {
const t = e.touches[0];
this.isPanning = true;
this.panStart = { x: t.clientX - this.offset.x, y: t.clientY - this.offset.y };
this.selectNode(null);
e.preventDefault();
}
}

onTouchMove(e) {
if (this.dragNode && e.touches.length === 1) {
const t = e.touches[0];
const rect = this.canvasWrap.getBoundingClientRect();
const sx = t.clientX - rect.left;
const sy = t.clientY - rect.top;
this.dragNode.x = (sx - this.offset.x - this.dragOffset.x) / this.scale;
this.dragNode.y = (sy - this.offset.y - this.dragOffset.y) / this.scale;
this.updateNodePosition(this.dragNode);
this.render();
e.preventDefault();
} else if (this.isPanning && e.touches.length === 1) {
const t = e.touches[0];
this.offset.x = t.clientX - this.panStart.x;
this.offset.y = t.clientY - this.panStart.y;
this.updateAllNodePositions();
this.render();
e.preventDefault();
} else if (this.connecting && e.touches.length === 1) {
const t = e.touches[0];
this.mouse.x = t.clientX;
this.mouse.y = t.clientY;
this.render();
e.preventDefault();
}
}

onTouchEnd(e) {
if (this.dragNode) {
this.net.broadcast({ type: 'node:move', data: { id: this.dragNode.id, x: this.dragNode.x, y: this.dragNode.y } });
this.dragNode = null;
this._scheduleSave();
}
if (this.isPanning) {
this.isPanning = false;
this._scheduleSave();
}
if (this.connecting && e.changedTouches.length > 0) {
const t = e.changedTouches[0];
const world = this.screenToWorld(t.clientX, t.clientY);
let target = null;
const hitRadius = NODE_VISUAL_RADIUS / this.scale + 5 / this.scale;
for (const n of this.nodes.values()) {
const dx = n.x - world.x, dy = n.y - world.y;
if (dxdx + dydy < hitRadius*hitRadius) { target = n; break; }
}
if (target && target.id !== this.connecting.fromId) {
this.promptEdgeLabel(this.connecting.fromId, target.id);
}
this.connecting = null;
this.render();
}
}

onWheel(e) {
e.preventDefault();
const rect = this.canvasWrap.getBoundingClientRect();
const mx = e.clientX - rect.left;
const my = e.clientY - rect.top;
const delta = e.deltaY > 0 ? 0.9 : 1.1;
const newScale = Math.max(0.2, Math.min(3, this.scale * delta));
const k = newScale / this.scale;
this.offset.x = mx - (mx - this.offset.x) * k;
this.offset.y = my - (my - this.offset.y) * k;
this.scale = newScale;
this.updateZoomLabel();
this.updateAllNodePositions();
this.render();
this._scheduleSave();
}

onDblClick(e) {
if (e.target.closest('.node')) return;
const world = this.screenToWorld(e.clientX, e.clientY);
this.promptAddNode(world.x, world.y);
}

onDrop(e) {
e.preventDefault();
const type = e.dataTransfer.getData('entity-type');
if (!type || !ENTITY_TYPES[type]) return;
const world = this.screenToWorld(e.clientX, e.clientY);
this.addNode(type, world.x, world.y);
}

onKey(e) {
if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') return;
if ((e.key === 'Delete' || e.key === 'Backspace') && this.selectedNode) {
this.deleteNode(this.selectedNode.id);
}
if (e.key === 'Escape') {
this.selectNode(null);
this.connecting = null;
this.hideContextMenu();
this.render();
}
}

startConnect(fromId, e) {
this.connecting = { fromId };
e.stopPropagation();
e.preventDefault();
}

// ---- Actions ----
zoomIn() {
this.scale = Math.min(3, this.scale * 1.2);
this.updateZoomLabel();
this.updateAllNodePositions();
this.render();
this._scheduleSave();
}
zoomOut() {
this.scale = Math.max(0.2, this.scale / 1.2);
this.updateZoomLabel();
this.updateAllNodePositions();
this.render();
this._scheduleSave();
}
updateZoomLabel() {
document.getElementById('zoomLevel').textContent = Math.round(this.scale * 100) + '%';
}

fitView() {
if (this.nodes.size === 0) return;
let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
for (const n of this.nodes.values()) {
minX = Math.min(minX, n.x); minY = Math.min(minY, n.y);
maxX = Math.max(maxX, n.x); maxY = Math.max(maxY, n.y);
}
const pad = 100 / this.scale;
const w = this.canvas.clientWidth;
const h = this.canvas.clientHeight;
const gw = maxX - minX + pad * 2;
const gh = maxY - minY + pad * 2;
this.scale = Math.min(w / gw, h / gh, 1.5);
const cx = (minX + maxX) / 2;
const cy = (minY + maxY) / 2;
this.offset.x = w / 2 - cx * this.scale;
this.offset.y = h / 2 - cy * this.scale;
this.updateZoomLabel();
this.updateAllNodePositions();
this.render();
this._scheduleSave();
}

resetView() {
this.offset = { x: 0, y: 0 };
this.scale = 1;
this.updateZoomLabel();
this.updateAllNodePositions();
this.render();
this._scheduleSave();
}

autoLayout() {
const nodes = [...this.nodes.values()];
if (nodes.length === 0) return;
const radius = Math.max(200, nodes.length * 60);
nodes.forEach((n, i) => {
const angle = (i / nodes.length) * Math.PI * 2;
n.x = Math.cos(angle) * radius;
n.y = Math.sin(angle) * radius;
this.updateNodePosition(n);
});
if (!this._remoteChange) {
for (const n of nodes) {
this.net.broadcast({ type: 'node:move', data: { id: n.id, x: n.x, y: n.y } });
}
}
this.render();
this.fitView();
}

clearAll() {
if (!confirm('Удалить все узлы и связи?')) return;
for (const n of this.nodes.values()) if (n.el) n.el.remove();
this.nodes.clear();
this.edges = [];
this.selectedNode = null;
this.nextId = 1;
this.updateStats();
this.render();
if (!this._remoteChange) {
this.net.broadcast({ type: 'sync', data: this.serialize() });
}
this._saveNow();
}

// ---- Persistence ----
serialize() {
return {
nodes: [...this.nodes.values()].map(n => ({
id: n.id, type: n.type, x: n.x, y: n.y, label: n.label
})),
edges: this.edges.map(e => ({ from: e.from, to: e.to, label: e.label })),
nextId: this.nextId
};
}

deserialize(data) {
for (const n of this.nodes.values()) if (n.el) n.el.remove();
this.nodes.clear();
this.edges = [];
for (const n of data.nodes) {
const node = { ...n };
this.nodes.set(node.id, node);
this.createNodeElement(node);
}
this.edges = data.edges || [];
this.nextId = data.nextId || 1;
this.updateStats();
this.updateAllNodePositions();
this.render();
}

manualSave() {
this._saveNow();
this.flash('Проект сохранён');
}

exportJSON() {
const blob = new Blob([JSON.stringify(this.serialize(), null, 2)], { type: 'application/json' });
const url = URL.createObjectURL(blob);
const a = document.createElement('a');
a.href = url;
a.download = 'osint-graph-' + Date.now() + '.json';
a.click();
URL.revokeObjectURL(url);
}

importJSON() {
const input = document.createElement('input');
input.type = 'file';
input.accept = 'application/json';
input.onchange = e => {
const f = e.target.files[0];
if (!f) return;
const r = new FileReader();
r.onload = ev => {
try {
this.deserialize(JSON.parse(ev.target.result));
this._saveNow();
this.flash('Импортировано и сохранено');
} catch (err) {
alert('Ошибка: ' + err.message);
}
};
r.readAsText(f);
};
input.click();
}

}

}
